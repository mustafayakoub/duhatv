1. surahs
─────────────────────────────────────────────────────────────────────────

CREATE TABLE surahs (
    surah_id INTEGER PRIMARY KEY,
    name_arabic TEXT NOT NULL,
    name_english TEXT NOT NULL,
    name_transliteration TEXT,
    ayah_count INTEGER NOT NULL,
    word_count INTEGER,
    letter_count INTEGER,
    revelation_type TEXT CHECK(revelation_type IN ('مكية', 'مدنية')),
    revelation_order INTEGER,
    juz_start INTEGER,
    juz_end INTEGER,
    page_start INTEGER,
    page_end INTEGER,
    description TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

مثال على البيانات:
┌─────┬──────────┬──────────┬───────────┬────────┬────────┬────────┐
│ ID  │ العربي   │ English  │ Translat  │ Ayahs  │ Type   │ Order  │
├─────┼──────────┼──────────┼───────────┼────────┼────────┼────────┤
│ 1   │ الفاتحة  │ Al-Fati  │ Al-Fatiha │ 7      │ مكية  │ 5      │

1. ayahs
─────────────────────────────────────────────────────────────────────────

CREATE TABLE ayahs (
    ayah_id INTEGER PRIMARY KEY AUTOINCREMENT,
    surah_id INTEGER NOT NULL,
    ayah_number INTEGER NOT NULL,
    text_uthmani TEXT NOT NULL,        -- نص بالرسم العثماني
    text_imlaei TEXT,                  -- نص بالرسم الإملائي
    text_simple TEXT,                  -- نص مبسط (بدون تشكيل)
    juz_number INTEGER,
    hizb_number INTEGER,
    manzil_number INTEGER,
    ruku_number INTEGER,
    page_number INTEGER,
    sajdah BOOLEAN DEFAULT 0,
    sajdah_type TEXT,
    UNIQUE(surah_id, ayah_number)
);

1. words
─────────────────────────────────────────────────────────────────────────

CREATE TABLE words (
    word_id INTEGER PRIMARY KEY AUTOINCREMENT,
    surah_id INTEGER NOT NULL,
    ayah_number INTEGER NOT NULL,
    word_position INTEGER NOT NULL,
    word_text TEXT NOT NULL,           -- الكلمة بالتشكيل
    word_simple TEXT NOT NULL,         -- بدون تشكيل
    word_uthmani TEXT,                 -- رسم عثماني
    root_text TEXT,
    root_id INTEGER,
    lemma TEXT,                        -- الأصل
    position_in_surah INTEGER,
    position_in_quran INTEGER,
    position_in_juz INTEGER,
    FOREIGN KEY (root_id) REFERENCES roots(root_id),
    UNIQUE(surah_id, ayah_number, word_position)
);

مثال:
┌────┬────┬────┬────┬─────────┬──────┬─────┬──────┬──────┐
│ ID │سورة│آية │ترتب│ نص      │بسيط  │جذر  │سورة  │قرآن  │
├────┼────┼────┼────┼─────────┼──────┼─────┼──────┼──────┤
│ 1  │ 1  │ 1  │ 1  │ بِسۡمِ   │ بسم  │ سمو │ 1    │ 1    │
│ 2  │ 1  │ 1  │ 2  │ ٱللَّهِ │ الله │ اله │ 2    │ 2    │
│ 3  │ 1  │ 1  │ 3  │ ٱلرَّحۡمَـٰنِ│الرحمن│رحم│ 3 │ 3    │

3. word_irab (الإعراب)
─────────────────────────────────────────────────────────────────────────

CREATE TABLE word_irab (
    irab_id INTEGER PRIMARY KEY AUTOINCREMENT,
    word_id INTEGER NOT NULL UNIQUE,
    irab_text TEXT NOT NULL,           -- النص الكامل

مثال استعلام شامل:
──────────────────────────────────────────────────────────────────────────

"أعطني معلومات كاملة عن كلمة معينة"

1. من SURAHS_DATABASE: اسم السورة، نوعها
2. من AYAHS_DATABASE: نص الآية كاملاً
3. من WORDS_DATABASE: الكلمة، الجذر، الإعراب، الصرف
4. من LETTERS_DATABASE: تحليل الحروف، أحكام التجويد


مثال استعلام بحث:

"ابحث عن كلمات من جذر 'رحم'"

1. WORDS_DATABASE: WHERE root_text = 'رحم'
   → نحصل على (word_id, surah_id, ayah_number)

2. SURAHS_DATABASE: WHERE surah_id IN (...)
   → نحصل على أسماء السور

3. AYAHS_DATABASE: WHERE surah_id + ayah_number IN (...)
   → نحصل على نصوص الآيات

4. العرض: سورة → آية → كلمة محددة
