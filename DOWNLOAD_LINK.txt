✨ مثال الاستخدام السريع

from qurankit import *

# البحث
search = QuranSearchComponent(database)
results = search.search('الرحمن')

# الشجرة
tree = QuranTreeWidget(QuranTreeComponent(database))

# العرض
display = QuranDisplayComponent()
html = display.format_ayah_html(ayah_data)

# الإشارات
bookmarks = QuranBookmarksComponent(database)
bookmarks.add_bookmark(1, 1, 'الفاتحة')

# المظاهر
theme = QuranThemeComponent('dark')
stylesheet = theme.apply_theme_to_stylesheet(widget)
