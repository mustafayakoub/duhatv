```
│         ↓                             ↓                         │

│  📝 WORDS_DATABASE.db      ←→  🔤 LETTERS_DATABASE.db         │
```

## 📚 القاعدة الأولى: SURAHS_DATABASE.db

### 🎯 الهدف
تخزين معلومات السور الكاملة مع جميع علومها

### 📋 الجداول

#### 1. `surahs` - الجدول الرئيسي

```sql
CREATE TABLE surahs (
    surah_id INTEGER PRIMARY KEY,
    surah_number INTEGER NOT NULL,

    -- الأسماء
    name_arabic TEXT NOT NULL,
    name_simple TEXT NOT NULL,
    name_english TEXT NOT NULL,
    name_transliteration TEXT,

    -- التصنيف
    revelation_type TEXT CHECK(revelation_type IN ('Meccan', 'Medinan')),
    revelation_order INTEGER,

    -- الإحصائيات
    ayah_count INTEGER NOT NULL,
    word_count INTEGER,
    letter_count INTEGER,

    -- الحروف المقطعة
    opening_letters TEXT,  -- مثل: "الم", "حم", NULL

    -- الموضوع الرئيسي
    main_theme_ar TEXT,
    main_theme_en TEXT,

    -- ملاحظات
    notes TEXT,

    -- التواريخ
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

**مثال على البيانات:**

| surah_id | name_arabic | name_english | revelation_type | ayah_count | opening_letters |
|----------|-------------|--------------|-----------------|------------|-----------------|
| 1 | الفاتحة | Al-Fatihah | Meccan | 7 | NULL |
| 2 | البقرة | Al-Baqarah | Medinan | 286 | الم |
| 3 | آل عمران | Ali 'Imran | Medinan | 200 | الم |

#### 2. `surah_names` - الأسماء المتعددة للسور

```sql
CREATE TABLE surah_names (
    name_id INTEGER PRIMARY KEY AUTOINCREMENT,
    surah_id INTEGER NOT NULL,
    name_arabic TEXT NOT NULL,
    name_type TEXT, -- 'primary', 'alternative'
    reference TEXT, -- مصدر الاسم

    FOREIGN KEY (surah_id) REFERENCES surahs(surah_id)
);
```

**مثال:**
```

name_type: "primary"

```

#### 3. `surah_themes` - موضوعات السور

```sql
CREATE TABLE surah_themes (
    theme_id INTEGER PRIMARY KEY AUTOINCREMENT,
    surah_id INTEGER NOT NULL,
    theme_arabic TEXT NOT NULL,
    theme_english TEXT,
    description TEXT,

    FOREIGN KEY (surah_id) REFERENCES surahs(surah_id)
);
```

#### 4. `revelation_occasions` - أسباب النزول

```sql
CREATE TABLE revelation_occasions (
    occasion_id INTEGER PRIMARY KEY AUTOINCREMENT,
    surah_id INTEGER,
    ayah_start INTEGER,
    ayah_end INTEGER,
    occasion_arabic TEXT NOT NULL,
    occasion_english TEXT,
    source TEXT, -- المصدر: ابن كثير، الطبري...
    authenticity TEXT, -- درجة الصحة

    FOREIGN KEY (surah_id) REFERENCES surahs(surah_id)
);
```

#### 5. `surah_statistics` - إحصائيات متقدمة

```sql
CREATE TABLE surah_statistics (
    stat_id INTEGER PRIMARY KEY AUTOINCREMENT,

    -- إحصائيات الكلمات

    -- إحصائيات الحروف
    unique_letters INTEGER,
    most_common_letter TEXT,

    -- التكرارات
    most_repeated_word TEXT,
    most_repeated_word_count INTEGER,

    FOREIGN KEY (surah_id) REFERENCES surahs(surah_id)
```

#### 1. `ayahs` - الجدول الرئيسي

```sql
CREATE TABLE ayahs (
    ayah_id INTEGER PRIMARY KEY,
    surah_id INTEGER NOT NULL,
    ayah_number INTEGER NOT NULL,

    -- النص
    ayah_text TEXT NOT NULL,           -- مع التشكيل الكامل
    ayah_text_simple TEXT NOT NULL,    -- بدون تشكيل
    ayah_text_uthmani TEXT,            -- الرسم العثماني

    -- الإحصائيات
    word_count INTEGER NOT NULL,
    letter_count INTEGER NOT NULL,

    -- التصنيف
    juz INTEGER,                       -- الجزء (1-30)
    hizb INTEGER,                      -- الحزب (1-60)
    page_mushaf INTEGER,               -- صفحة المصحف

    -- السجدة
    sajdah_type TEXT,                  -- 'recommended', 'obligatory', NULL

    -- ملاحظات
    notes TEXT,

    -- الفهرسة
    search_text TEXT,                  -- نص بدون تشكيل للبحث

    UNIQUE(surah_id, ayah_number)
);

-- فهرس للبحث السريع
CREATE INDEX idx_ayahs_search ON ayahs(search_text);
CREATE INDEX idx_ayahs_surah ON ayahs(surah_id);
```

```sql
CREATE TABLE tafsir (
```

```sql
CREATE TABLE translations (
    translation_id INTEGER PRIMARY KEY AUTOINCREMENT,
```

## 📝 القاعدة الثالثة: WORDS_DATABASE.db

### 🎯 الهدف
تخزين جميع كلمات القرآن مع الإعراب والتحليل الصرفي

### 📋 الجداول

#### 1. `words` - الجدول الرئيسي

```sql
CREATE TABLE words (
    word_id INTEGER PRIMARY KEY,
    surah_id INTEGER NOT NULL,
    ayah_id INTEGER NOT NULL,
    word_position INTEGER NOT NULL,    -- موقع الكلمة في الآية (1, 2, 3...)

    -- النص
    word_text TEXT NOT NULL,           -- مع التشكيل
    word_simple TEXT NOT NULL,         -- بدون تشكيل
    word_uthmani TEXT,                 -- الرسم العثماني

    -- التحليل اللغوي
    root TEXT,                         -- الجذر: "كتب"
    root_type TEXT,                    -- ثلاثي، رباعي...

    -- التصنيف
    word_type TEXT,                    -- 'اسم', 'فعل', 'حرف'
    word_category TEXT,                -- تفصيل: 'اسم فاعل', 'فعل ماضي'...

    -- الإحصائيات
    letter_count INTEGER,

    UNIQUE(ayah_id, word_position)
);

CREATE INDEX idx_words_ayah ON words(ayah_id);
CREATE INDEX idx_words_root ON words(root);
CREATE INDEX idx_words_simple ON words(word_simple);
```

**مثال:**

```
word_id: 1

ayah_id: 1
word_position: 1
word_text: "بِسْمِ"
word_simple: "بسم"

word_type: "اسم"
```

#### 2. `irab` - الإعراب

```sql
CREATE TABLE irab (
    irab_id INTEGER PRIMARY KEY AUTOINCREMENT,
    word_id INTEGER NOT NULL,

    -- الإعراب
    irab_text TEXT NOT NULL,           -- "جار ومجرور متعلق بمحذوف..."
    irab_position TEXT,                -- 'مبتدأ', 'خبر', 'فاعل'...
    irab_case TEXT,                    -- 'مرفوع', 'منصوب', 'مجرور', 'مجزوم'
    irab_sign TEXT,                    -- 'الضمة', 'الفتحة', 'الكسرة'...

    -- المصدر
    source TEXT,                       -- "إعراب القرآن للدرويش"
    notes TEXT,

    FOREIGN KEY (word_id) REFERENCES words(word_id)
);

CREATE INDEX idx_irab_word ON irab(word_id);
```

**مثال:**

```
word_id: 1

```

#### 3. `morphology` - التحليل الصرفي

```sql
CREATE TABLE morphology (
    morph_id INTEGER PRIMARY KEY AUTOINCREMENT,
    word_id INTEGER NOT NULL,

    -- الصيغة
    form TEXT,                         -- 'مفرد', 'مثنى', 'جمع'
    gender TEXT,                       -- 'مذكر', 'مؤنث'
    person TEXT,                       -- 'متكلم', 'مخاطب', 'غائب'
    tense TEXT,                        -- 'ماضي', 'مضارع', 'أمر' (للأفعال)

    -- الصيغة الصرفية
    pattern TEXT,                      -- 'فَعَلَ', 'فاعل', 'مفعول'...

    -- ملاحظات
    notes TEXT,

    FOREIGN KEY (word_id) REFERENCES words(word_id)
);
```

#### 4. `word_meanings` - معاني الكلمات

```sql
CREATE TABLE word_meanings (
```

```sql
CREATE TABLE word_statistics (
```

```sql
CREATE TABLE letter_positions (
    position_id INTEGER PRIMARY KEY AUTOINCREMENT,
```

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
استيراد البيانات من Excel
"""

import pandas as pd
import sqlite3
import argparse

def import_surahs(excel_file, db_file):
    """استيراد السور من Excel"""
    # قراءة Sheet 1: Basic Info
    df = pd.read_excel(excel_file, sheet_name='Basic Info')

    db = sqlite3.connect(db_file)

    for _, row in df.iterrows():
        db.execute("""
            INSERT INTO surahs (
                surah_id, name_arabic, name_english,
                revelation_type, ayah_count, revelation_order,
                opening_letters, main_theme_ar
            ) VALUES (?, ?, ?, ?, ?, ?, ?, ?)
        """, (
            row['surah_id'],
            row['name_arabic'],
            row['name_english'],
            row['revelation_type'],
            row['ayah_count'],
            row['revelation_order'],
            row['opening_letters'],
            row['main_theme_ar']
        ))

    db.commit()
    db.close()
    print(f"✅ تم استيراد {len(df)} سورة")

if __name__ == "__main__":
```

```python
import sqlite3

def verify_surahs():
    """التحقق من قاعدة السور"""
    db = sqlite3.connect('SURAHS_DATABASE.db')

    # التحقق من العدد
    count = db.execute("SELECT COUNT(*) FROM surahs").fetchone()[0]
    assert count == 114, f"❌ عدد السور خاطئ: {count} (المتوقع: 114)"
    print("✅ عدد السور صحيح: 114")

    # التحقق من الأسماء
    nulls = db.execute("""
        SELECT COUNT(*) FROM surahs
        WHERE name_arabic IS NULL OR name_arabic = ''
    """).fetchone()[0]
    assert nulls == 0, "❌ يوجد سور بدون أسماء"
    print("✅ جميع السور لها أسماء")

    db.close()

def verify_words():
    """التحقق من قاعدة الكلمات"""
    db = sqlite3.connect('WORDS_DATABASE.db')

    count = db.execute("SELECT COUNT(*) FROM words").fetchone()[0]
    expected = 77432
    assert count == expected, f"❌ عدد الكلمات خاطئ: {count} (المتوقع: {expected})"
    print(f"✅ عدد الكلمات صحيح: {count:,}")

    db.close()

if __name__ == "__main__":
    print("🔍 بدء التحقق من البيانات...\n")

    verify_surahs()
    verify_words()

    print("\n✅ جميع الفحوصات نجحت!")
```
